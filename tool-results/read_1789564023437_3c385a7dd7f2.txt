    70→interface ShoppingEntry {
    71→  key: string
    72→  label: string
    73→  count: number
    74→}
    75→
    76→/** Agrupa los ingredientes de todas las recetas del plan. */
    77→function buildShoppingList(items: PlanItemData[]): ShoppingEntry[] {
    78→  const map = new Map<string, ShoppingEntry>()
    79→  for (const item of items) {
    80→    for (const raw of item.recipe.ingredients) {
    81→      const label = raw.replace(/\s+/g, ' ').trim()
    82→      if (!label) continue
    83→      const key = label.toLowerCase()
    84→      const existing = map.get(key)
    85→      if (existing) existing.count += 1
    86→      else map.set(key, { key, label, count: 1 })
    87→    }
    88→  }
    89→  return Array.from(map.values()).sort(
    90→    (a, b) => b.count - a.count || a.label.localeCompare(b.label, 'es')
    91→  )
    92→}

   120→      setLoading(false)

   135→
   136→  function itemAt(day: number, slot: PlanSlot): PlanItemData | undefined {
   137→    return items?.find((it) => it.day === day && it.slot === slot)
   138→  }
   139→
   140→  async function assign(recipe: PlanRecipeSummary | RecipeCardData) {
   141→    if (!picker) return
   142→    try {
   143→      const d = await api<{ item: PlanItemData }>(
   144→        '/api/plan',
   145→        jsonBody('PUT', { day: picker.day, slot: picker.slot, recipeId: recipe.id })
   146→      )
   147→      setItems((prev) => {
   148→        const rest = (prev ?? []).filter(
   149→          (it) => !(it.day === picker.day && it.slot === picker.slot)
   150→        )
   151→        return [...rest, d.item]
   152→      })
   153→      toast.success(`«${recipe.title}» añadida al ${PLAN_DAY_NAMES[picker.day].toLowerCase()} (${PLAN_SLOTS.find((s) => s.value === picker.slot)?.label.toLowerCase()}).`)
   154→      setPicker(null)
   155→    } catch (err) {
   156→      toast.error(err instanceof Error ? err.message : 'No se pudo añadir la receta al plan.')
   157→    }
   158→  }
   159→
   160→  async function removeItem(id: string) {
   161→    try {
   162→      await api(`/api/plan/${id}`, jsonBody('DELETE', {}))
   163→      setItems((prev) => (prev ?? []).filter((it) => it.id !== id))
   164→      toast.success('Receta quitada del plan.')
   165→    } catch (err) {
   166→      toast.error(err instanceof Error ? err.message : 'No se pudo quitar la receta.')
   167→    }
   168→  }
   169→
   170→  async function clearAll() {
   171→    try {
   172→      await api('/api/plan', jsonBody('DELETE', {}))
   173→      setItems([])
   174→      setChecked(new Set())
   175→      toast.success('Plan semanal vaciado.')
   176→    } catch (err) {
   177→      toast.error(err instanceof Error ? err.message : 'No se pudo vaciar el plan.')
   178→    }
   179→  }

   181→  /** Copia todas las franjas planificadas de un día a otro (sobrescribe el destino). */
   182→  async function copyDay(from: number, to: number) {
   183→    const source = PLAN_SLOTS.map(({ value }) => itemAt(from, value)).filter(
   184→      (it): it is PlanItemData => !!it
   185→    )
   186→    setCopySource(null)
   187→    if (source.length === 0) return
   188→    try {
   189→      await Promise.all(
   190→        source.map((it) =>
   191→          api('/api/plan', jsonBody('PUT', { day: to, slot: it.slot, recipeId: it.recipe.id }))
   192→        )
   193→      )
   194→      await fetchPlan()
   195→      toast.success(
   196→        `Menú del ${PLAN_DAY_NAMES[from].toLowerCase()} copiado al ${PLAN_DAY_NAMES[to].toLowerCase()} (${source.length} ${source.length === 1 ? 'receta' : 'recetas'}).`
   197→      )
   198→    } catch (err) {
   199→      toast.error(err instanceof Error ? err.message : 'No se pudo copiar el menú del día.')
   200→    }
   201→  }

   203→  /** Descarga la lista de la compra como CSV compatible con Excel-es (BOM + ";"). */
   204→  function exportCsv() {
   205→    if (shoppingList.length === 0) return
   206→    const esc = (v: string) => `"${v.replace(/"/g, '""')}"`
   207→    const rows = ['Ingrediente;Raciones que lo usan']
   208→    for (const e of shoppingList) rows.push(`${esc(e.label)};${e.count}`)
   209→    const csv = '\uFEFF' + rows.join('\r\n')
   210→    const url = URL.createObjectURL(new Blob([csv], { type: 'text/csv;charset=utf-8' }))
   211→    const a = document.createElement('a')
   212→    a.href = url
   213→    a.download = `lista-compra-simbiosis-${new Date().toISOString().slice(0, 10)}.csv`
   214→    a.click()
   215→    URL.revokeObjectURL(url)
   216→    toast.success('Lista de la compra exportada a CSV.')
   217→  }

   521→                <div className="flex items-center justify-between gap-2 print:hidden">
   522→                  <span className="text-xs text-muted-foreground">
   523→                    {checked.size} de {shoppingList.length} marcados
   524→                  </span>
   525→                  <div className="flex items-center gap-1.5">
   526→                    <Button
   527→                      variant="outline"
   528→                      size="sm"
   529→                      className="min-h-9"
   530→                      onClick={exportCsv}
   531→                      aria-label="Exportar la lista de la compra en CSV"
   532→                    >
   533→                      <Download aria-hidden="true" className="size-4" />
   534→                      CSV
   535→                    </Button>
   536→                    <Button variant="outline" size="sm" className="min-h-9" onClick={() => void copyList()}>
   537→                      <ClipboardCopy aria-hidden="true" className="size-4" />
   538→                      Copiar lista
   539→                    </Button>
   540→                  </div>
   541→                </div>
   542→              </>
   543→            )}
   544→          </CardContent>
   545→        </Card>
   546→      </div>
   547→
   548→      {/* Selector de receta */}
   549→      <RecipePickerDialog
   550→        picker={picker}
   551→        onClose={() => setPicker(null)}
   552→        onPick={(recipe) => void assign(recipe)}
   553→      />
   554→    </div>
   555→  )
   556→}

   558→/** Diálogo para buscar y elegir la receta que ocupará un hueco del plan. */
   559→function RecipePickerDialog({
   560→  picker,
   561→  onClose,
   562→  onPick,
   563→}: {
   564→  picker: { day: number; slot: PlanSlot } | null
   565→  onClose: () => void
   566→  onPick: (recipe: PlanRecipeSummary | RecipeCardData) => void
   567→}) {
   568→  const [query, setQuery] = useState('')
   569→  const [results, setResults] = useState<PlanRecipeSummary[] | null>(null)
   570→  const [loading, setLoading] = useState(false)
   571→
   572→  useEffect(() => {
   573→    if (!picker) return
   574→    let active = true
   575→    const t = setTimeout(() => {
   576→      setLoading(true)
   577→      api<{ recipes: (RecipeCardData & { ingredients?: string[] })[] }>(
   578→        `/api/recipes?q=${encodeURIComponent(query)}`
   579→      )
   580→        .then((d) => {
   581→          if (!active) return
   582→          setResults(
   583→            d.recipes.map((r) => ({
   584→              id: r.id,
   585→              title: r.title,
   586→              image: r.image,
   587→              category: r.category,
   588→              ingredients: r.ingredients ?? [],
   589→              prepTime: r.prepTime,
   590→              servings: r.servings,
   591→              author: r.author ?? { id: '', name: 'Comunidad', role: 'PATIENT' },
   592→            }))
   593→          )
   594→        })
   595→        .catch(() => active && setResults([]))
   596→        .finally(() => active && setLoading(false))
   597→    }, 250)
   598→    return () => {
   599→      active = false
   600→      clearTimeout(t)
   601→    }
   602→  }, [picker, query])
   603→
   604→  function handleClose() {
   605→    setQuery('')
   606→    setResults(null)
   607→    onClose()
   608→  }
   609→
   610→  const slotLabel = picker ? PLAN_SLOTS.find((s) => s.value === picker.slot)?.label : ''
   611→  const dayLabel = picker ? PLAN_DAY_NAMES[picker.day] : ''
   612→
   613→  return (
   614→    <Dialog open={!!picker} onOpenChange={(open) => !open && handleClose()}>
   615→      <DialogContent className="max-h-[85vh] overflow-hidden sm:max-w-lg">
   616→        <DialogHeader>
   617→          <DialogTitle>Añadir receta · {dayLabel} · {slotLabel}</DialogTitle>
   618→          <DialogDescription>
   619→            Busca una receta de la comunidad para este hueco del plan. Sustituirá la actual si la hay.
   620→          </DialogDescription>
   621→        </DialogHeader>
   622→        <Input
   623→          placeholder="Buscar recetas por título o descripción…"
   624→          value={query}
   625→          onChange={(e) => setQuery(e.target.value)}
   626→          aria-label="Buscar recetas para el plan"
   627→        />
   628→        <div className="max-h-80 space-y-1.5 overflow-y-auto pr-1">
   629→          {results === null ? (
   630→            <div className="flex items-center justify-center py-8">
   631→              <Loader2 className="size-5 animate-spin text-primary" aria-hidden="true" />
   632→            </div>
   633→          ) : (results ?? []).length === 0 ? (
   634→            <p className="flex flex-col items-center gap-2 py-8 text-center text-sm text-muted-foreground">
   635→              <UtensilsCrossed aria-hidden="true" className="size-6" />
   636→              No hay recetas que coincidan con la búsqueda.
   637→            </p>
   638→          ) : (
   639→            (results ?? []).map((recipe) => (
   640→              <button
   641→                key={recipe.id}
   642→                type="button"
   643→                className="flex w-full items-center gap-3 rounded-xl border p-2 text-left outline-none transition-colors hover:border-primary/40 hover:bg-accent focus-visible:ring-2 focus-visible:ring-ring"
   644→                onClick={() => onPick(recipe)}
   645→                aria-label={`Añadir ${recipe.title} al plan`}
   646→              >
   647→                <span className="relative size-12 shrink-0 overflow-hidden rounded-lg">
   648→                  <ImageWithFallback src={recipe.image} alt="" sizes="48px" icon="none" />
   649→                </span>
   650→                <span className="min-w-0 flex-1">
   651→                  <span className="block truncate text-sm font-medium">{recipe.title}</span>
   652→                  <span className="block text-xs text-muted-foreground">
   653→                    {recipe.category} · {recipe.prepTime} min · {recipe.servings}{' '}
   654→                    {recipe.servings === 1 ? 'ración' : 'raciones'}
   655→                  </span>
   656→                </span>
   657→                <Plus aria-hidden="true" className="size-4 shrink-0 text-primary" />
   658→              </button>
   659→            ))
   660→          )}
   661→        </div>
   662→      </DialogContent>
   663→    </Dialog>
   664→  )
   665→}
