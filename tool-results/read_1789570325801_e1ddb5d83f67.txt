    94→/** Agrupa los ingredientes de todas las recetas del plan. */
    95→function buildShoppingList(items: PlanItemData[]): ShoppingEntry[] {
    96→  const map = new Map<string, ShoppingEntry>()
    97→  for (const item of items) {
    98→    for (const raw of item.recipe.ingredients) {
    99→      const label = raw.replace(/\s+/g, ' ').trim()
   100→      if (!label) continue
   101→      const key = label.toLowerCase()
   102→      const existing = map.get(key)
   103→      if (existing) existing.count += 1
   104→      else map.set(key, { key, label, count: 1 })
   105→    }
   106→  }
   107→  return Array.from(map.values()).sort(
   108→    (a, b) => b.count - a.count || a.label.localeCompare(b.label, 'es')
   109→  )
   110→}

   306→  /** Descarga la lista de la compra como CSV compatible con Excel-es (BOM + ";"). */
   307→  function exportCsv() {
   308→    if (shoppingList.length === 0) return
   309→    const esc = (v: string) => `"${v.replace(/"/g, '""')}"`
   310→    const rows = ['Ingrediente;Raciones que lo usan']
   311→    for (const e of shoppingList) rows.push(`${esc(e.label)};${e.count}`)
   312→    const csv = '\uFEFF' + rows.join('\r\n')
   313→    const url = URL.createObjectURL(new Blob([csv], { type: 'text/csv;charset=utf-8' }))
   314→    const a = document.createElement('a')
   315→    a.href = url
   316→    a.download = `lista-compra-simbiosis-${new Date().toISOString().slice(0, 10)}.csv`
   317→    a.click()
   318→    URL.revokeObjectURL(url)
   319→    toast.success('Lista de la compra exportada a CSV.')
   320→  }
